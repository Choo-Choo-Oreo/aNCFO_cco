on_actions = {
	on_startup = {
		effect = { # Scripted Triggers Initialization
			set_global_flag = is_OCfA_enabled

			set_variable = { global.OCfA_quarterly_counter = 3 }
			log = "GLOBAL LOG: Quarterly Pulse Triggered on [GetDateText]. Variable Reset to 3."
			
			set_variable = { global.OCfA_yearly_counter = 12 }
			log = "GLOBAL LOG: Yearly Pulse Triggered on [GetDateText]. Variable Reset to 12."

			IF = {
				LIMIT = {
					has_game_rule = { rule = OCfA_mobilization_scaling_rule option = MOB_ON }
				}
				set_global_flag = is_OCfA_mobilization_scaling_rule_enabled
			}

			set_global_flag = { flag = OCfA_prevent_demobilization days = 730 value = 1 }

			every_possible_country = {
				IF = {
					LIMIT = {
						NOT = { has_dynamic_modifier = { modifier = OCfA_opereation_capacity_dynamic_modifier } }
					}
					add_dynamic_modifier = { modifier = OCfA_opereation_capacity_dynamic_modifier }
				}
				OCfA_calculate_battalion_capacity_formula = yes
			}
		}
	}

	on_weekly = { # on_monthly is to slow for the rapid change of unit's
		effect = {
			IF = {
				LIMIT = {
					NOT = { has_dynamic_modifier = { modifier = OCfA_opereation_capacity_dynamic_modifier } }
				}
				add_dynamic_modifier = { modifier = OCfA_opereation_capacity_dynamic_modifier }
			}
			OCfA_calculate_battalion_capacity_formula = yes
		}
	}
	# Change the `ZZZ` to any confirmed scope.
	on_monthly = {
		effect = {
			IF = {
				LIMIT = { NOT = { has_global_flag = OCfA_global_monthly_done } }
				set_global_flag = { flag = OCfA_global_monthly_done days = 1 value = 1 }

				log = "OQLY / OYLY fire on_monthly_TAG"
				subtract_from_variable = { global.OCfA_quarterly_counter = 1 }
				subtract_from_variable = { global.OCfA_yearly_counter = 1 }

				# --- ON_QUARTERLY (OCfA) ---
				IF = {
					LIMIT = { check_variable = { global.OCfA_quarterly_counter = 0 } }
					log = "OCfA fire OCfA_generate_csv_battalion_capacity_data = yes"
					OCfA_generate_csv_battalion_capacity_data = yes
				}

				# --- ON_YEARLY (OCfA) ---

				# --- ON_QUARTERLY (Country Scoped) ---
				IF = {
					LIMIT = { check_variable = { global.OCfA_quarterly_counter = 0 } }
					
					# --- ON_QUARTERLY ---
					log = "GLOBAL LOG: Quarterly Pulse Triggered on [GetDateText]. Variable Reset to 3."
					
					# --- RESET ---
					set_variable = { global.OCfA_quarterly_counter = 3 }
				}
				
				# --- ON_YEARLY (Country Scoped) ---
				IF = {
					LIMIT = { check_variable = { global.OCfA_yearly_counter = 0 } }
					
					# --- ON_YEARLY ---
					log = "GLOBAL LOG: Yearly Pulse Triggered on [GetDateText]. Variable Reset to 12."
					
					# --- RESET ---
					set_variable = { global.OCfA_yearly_counter = 12 }
				}
			}
		}
	}

	on_peace = {
		effect = {
			log = "[GetDateText]: [Root.GetName]: on_peace"
			THIS = {
				OCfA_calculate_battalion_capacity_formula = yes
				set_country_flag = OCfA_allow_demobilization
			}
		}
	}
}
