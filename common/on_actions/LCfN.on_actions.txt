on_actions = {
	on_startup = {
		effect = { # Scripted Triggers Initialization
			set_global_flag = is_LCfN_enabled

			set_variable = { global.LCfN_quarterly_counter = 3 }
			log = "GLOBAL LOG: Quarterly Pulse Triggered on [GetDateText]. Variable Reset to 3."
			
			set_variable = { global.LCfN_yearly_counter = 12 }
			log = "GLOBAL LOG: Yearly Pulse Triggered on [GetDateText]. Variable Reset to 12."

			every_possible_country = {
				IF = {
					LIMIT = {
						NOT = { has_dynamic_modifier = { modifier = LCfN_naval_capacity_dynamic_modifier } }
					}
					add_dynamic_modifier = { modifier = LCfN_naval_capacity_dynamic_modifier }
				}
				LCfN_calculate_naval_capacity_logistics_formula = yes
			}
		}
	}

	on_monthly = {
		effect = {
			IF = {
				LIMIT = {
					NOT = { has_dynamic_modifier = { modifier = LCfN_naval_capacity_dynamic_modifier } }
				}
				add_dynamic_modifier = { modifier = LCfN_naval_capacity_dynamic_modifier }
			}
			LCfN_calculate_naval_capacity_logistics_formula = yes
			LCfN_monthly_convoy_simulation = yes
		}
		# Change the `ZZZ` to any confirmed scope.
		effect = {
			IF = {
				LIMIT = { NOT = { has_global_flag = LCfN_global_monthly_done } }
				set_global_flag = { flag = LCfN_global_monthly_done days = 1 value = 1 }

				log = "OQLY / OYLY fire on_monthly_TAG"
				subtract_from_variable = { global.LCfN_quarterly_counter = 1 }
				subtract_from_variable = { global.LCfN_yearly_counter = 1 }

				# --- ON_QUARTERLY (LCfN) ---
				IF = {
					LIMIT = { check_variable = { global.LCfN_quarterly_counter = 0 } }
					log = "LCfN fire LCfN_generate_csv_naval_capacity_data = yes"
					LCfN_generate_csv_naval_capacity_data = yes
				}

				# --- ON_YEARLY (LCfN) ---

				# --- ON_QUARTERLY (Country Scoped) ---
				IF = {
					LIMIT = { check_variable = { global.LCfN_quarterly_counter = 0 } }
					
					# --- ON_QUARTERLY ---
					log = "GLOBAL LOG: Quarterly Pulse Triggered on [GetDateText]. Variable Reset to 3."
					
					# --- RESET ---
					set_variable = { global.LCfN_quarterly_counter = 3 }
				}
				
				# --- ON_YEARLY (Country Scoped) ---
				IF = {
					LIMIT = { check_variable = { global.LCfN_yearly_counter = 0 } }
					
					# --- ON_YEARLY ---
					log = "GLOBAL LOG: Yearly Pulse Triggered on [GetDateText]. Variable Reset to 12."
					
					# --- RESET ---
					set_variable = { global.LCfN_yearly_counter = 12 }
				}
			}
		}
	}

	on_peace = {
		effect = {
			log = "[GetDateText]: [Root.GetName]: on_peace"
			THIS = { LCfN_calculate_naval_capacity_logistics_formula = yes }
		}
	}
}
