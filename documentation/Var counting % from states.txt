count_proportion_owned_heritage = {
    hidden_trigger = {
        all_of_scopes = {
            array = global.SEH_heritage_states_@var:temp_heritage_check

    divide_temp_variable = { proportion = global.SEH_heritage_states_@var:temp_heritage_check^num }

    subtract_from_temp_variable = { proportion_display = 1 }

    custom_trigger_tooltip = {
        tooltip = count_proportion_owned_heritage_tt
        check_variable = { var = proportion value = temp_min_proportion compare = greater_than_or_equals }
    }
}
